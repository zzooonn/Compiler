void SumofArray(int arr[], int size)
{
    int sum = 0;
    int i;
    for (i = 0; i < size; i = i + 1)
    {
        sum = sum + arr[i];
    }
    printf("Sum value: %d\n", sum);


    if (sum > 200 || sum < 10)
    {
        printf("Sum is greater than 200\n");
    }
    else if (sum > 100 && sum < 199)
    {
        printf("Sum is greater than 100 but less than or equal to 200\n");
    }
    else
    {
        printf("Sum is less than or equal to 100\n");
    }
}


void findMax(int arr[], int size)
{
    int max = arr[0];
    int i;
    for (i = 1; i < size; i = i + 1)
    {
        if (arr[i] > max)
        {
            max = arr[i];
        }
    }
    printf("Max value: %d\n", max);
}
