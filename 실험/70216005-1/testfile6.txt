int main()
{
    int i = 0;

    for (i = 1; i <= 10; i = i + 1)
    {
        if (i % 2 == 0) 
        {
            continue;  
        }
        printf("%d\n", i);  
    }


    for (; i < 10; i = i + 1)
    { 
        printf("%d\n", i);
    }

    for (; i < 10;) 
    {  
        printf("%d\n", i);
        i = i + 1; 
    }

    for (;;) 
    {  
        if (i >= 10)
        {
            break;  
        }
        printf("%d\n", i);
        i = i + 1;  
    }


    return 0;
}
